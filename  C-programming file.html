1. Area of Circle & Circumference

#include <stdio.h>

int main() {
    float r, area, circumference;

    printf("Enter radius: ");
    scanf("%f", &r);

    area = 3.14 * r * r;
    circumference = 2 * 3.14 * r;

    printf("Area = %.2f\n", area);
    printf("Circumference = %.2f\n", circumference);

    return 0;
}

4. Quadratic Equations

#include <stdio.h>
#include <math.h>

int main() {
    float a, b, c, d, root1, root2;

    printf("Enter a, b and c: ");
    scanf("%f %f %f", &a, &b, &c);

    d = b * b - 4 * a * c;

    if (d > 0) {
        root1 = (-b + sqrt(d)) / (2 * a);
        root2 = (-b - sqrt(d)) / (2 * a);

        printf("Root 1 = %.2f\n", root1);
        printf("Root 2 = %.2f", root2);
    }
    else if (d == 0) {
        root1 = -b / (2 * a);
        printf("Both roots = %.2f", root1);
    }
    else {
        printf("Roots are imaginary");
    }

    return 0;
}

5. GCD

#include <stdio.h>

int main() {
    int a, b, rem;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    while (b != 0) {
        rem = a % b;
        a = b;
        b = rem;
    }

    printf("GCD = %d", a);

    return 0;
}

6. Prime Number

#include <stdio.h>

int main() {
    int n, i, flag = 0;

    printf("Enter a number: ");
    scanf("%d", &n);

    if (n <= 1)
        flag = 1;

    for (i = 2; i <= n / 2; i++) {
        if (n % i == 0) {
            flag = 1;
            break;
        }
    }

    if (flag == 0)
        printf("%d is Prime", n);
    else
        printf("%d is Not Prime", n);

    return 0;
}

7. Reverse Number

#include <stdio.h>

int main() {
    int n, reverse = 0, digit;

    printf("Enter a number: ");
    scanf("%d", &n);

    while (n != 0) {
        digit = n % 10;
        reverse = reverse * 10 + digit;
        n = n / 10;
    }

    printf("Reverse = %d", reverse);

    return 0;
}
